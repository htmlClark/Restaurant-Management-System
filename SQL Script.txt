CREATE DATABASE restaurant_management_system;

SET FOREIGN_KEY_CHECKS = 0;

CREATE TABLE role (
-> role_id INT AUTO_INCREMENT PRIMARY KEY,
-> role VARCHAR(20) NOT NULL UNIQUE);

CREATE TABLE userlist (
-> user_id INT AUTO_INCREMENT PRIMARY KEY,
-> role_id INT NOT NULL,
-> name VARCHAR(150) NOT NULL,
-> password VARCHAR(150) NOT NULL,
-> FOREIGN KEY (role_id) REFERENCES role(role_id) ON DELETE CASCADE ON UPDATE CASCADE);

CREATE TABLE login (
-> login_id INT AUTO_INCREMENT PRIMARY KEY,
-> user_id INT NOT NULL,
-> time_login DATETIME,
-> time_logout DATETIME,
-> FOREIGN KEY (user_id) REFERENCES userlist(user_id) ON DELETE CASCADE ON UPDATE CASCADE);

CREATE TABLE dish_list (
-> dish_id INT AUTO_INCREMENT PRIMARY KEY,
-> dish_name VARCHAR(100) NOT NULL,
-> dish_category VARCHAR(50) NOT NULL,
-> price DECIMAL(10,2) NOT NULL);

CREATE TABLE inventory (
-> inv_id VARCHAR(10) PRIMARY KEY,
-> ingredient_name VARCHAR(100) NOT NULL,
-> quantity DECIMAL(10,2) NOT NULL,
-> dish_category VARCHAR(50) NOT NULL,
-> measurement DECIMAL(10,2) NOT NULL);

CREATE TABLE orders (
-> order_id INT AUTO_INCREMENT PRIMARY KEY,
-> dish_id INT NOT NULL,
-> order_date DATE NOT NULL,
-> order_time TIME NOT NULL,
-> status VARCHAR(50) NOT NULL,
-> FOREIGN KEY (dish_id) REFERENCES dish_list(dish_id) ON DELETE CASCADE ON UPDATE CASCADE); 
CREATE TABLE order_items (
-> order_item_id INT AUTO_INCREMENT PRIMARY KEY,
-> order_id INT NOT NULL,
-> dish_id INT NOT NULL,
-> payment_id INT NOT NULL,
-> quantity INT NOT NULL,
-> price DECIMAL(10,2) NOT NULL,
-> FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE ON UPDATE CASCADE,
-> FOREIGN KEY (dish_id) REFERENCES dish_list(dish_id) ON DELETE CASCADE ON UPDATE CASCADE,
-> FOREIGN KEY (payment_id) REFERENCES payment_method(payment_id) ON DELETE CASCADE ON UPDATE CASCADE);

CREATE TABLE recipe (
-> recipe_id INT AUTO_INCREMENT PRIMARY KEY,
-> inv_id VARCHAR(10) NOT NULL,
-> dish_id INT NOT NULL,
-> quantity INT NOT NULL,
-> FOREIGN KEY (inv_id) REFERENCES inventory(inv_id) ON DELETE CASCADE ON UPDATE CASCADE,
-> FOREIGN KEY (dish_id) REFERENCES dish_list(dish_id) ON DELETE CASCADE ON UPDATE CASCADE);

CREATE TABLE wastelogs (
-> waste_id INT AUTO_INCREMENT PRIMARY KEY,
-> inv_id VARCHAR(10) NOT NULL,
-> user_id INT NOT NULL,
-> quantity DECIMAL(10,2) NOT NULL,
-> reason VARCHAR(100) NOT NULL,
-> remarks VARCHAR(300) NOT NULL,
-> FOREIGN KEY (inv_id) REFERENCES inventory(inv_id) ON DELETE CASCADE ON UPDATE CASCADE,
-> FOREIGN KEY (user_id) REFERENCES userlist(user_id) ON DELETE CASCADE ON UPDATE CASCADE);

CREATE TABLE payment_method (
-> payment_id INT AUTO_INCREMENT PRIMARY KEY,
-> payment_method VARCHAR(100) NOT NULL);

CREATE TABLE delivery (
-> delivery_id VARCHAR(10) PRIMARY KEY,
-> inv_id VARCHAR(10) NOT NULL,
-> expiration_date DATE NOT NULL,
-> delivery_date DATE NOT NULL,
-> delivery_time TIME NOT NULL,
-> courier VARCHAR(100) NOT NULL,
-> FOREIGN KEY (inv_id) REFERENCES inventory(inv_id) ON DELETE CASCADE ON UPDATE CASCADE);
